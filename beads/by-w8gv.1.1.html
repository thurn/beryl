just scale: cloned-card scale rehearsal harness, pytest, wired in justfile

Description

Build just scale (tools/src/beryl_tools, wired in the justfile): it generates a scratch Cargo workspace under $BERYL_CACHE (never in the tree), builds it, and prints a table of cold cargo check, release build, rebuild after a one-card edit, and the largest compiler process's peak memory.

  • Input: --cards N (default 50,000) and --crates K. Definitions are clones of the real card files under rust/crates/beryl_cards/src/<color>/ with the card name made unique, cycled until N is reached, so the workload is implemented cards with their scenarios, not stubs. Path dependencies point at the checkout's crates.
  • The final binary must reference every card (iterate the tables through std::hint::black_box) or the linker strips them and the release numbers mean nothing.
  • Measure with /usr/bin/time -l (wall and maximum resident set size). Warm the engine crates in each profile first so only the card crates are timed.
  • Report only: budgets are asserted from milestone 7 (--assert, plan Acceptance: check 60 s, release 180 s, one-card rebuild 2 s, 4 GiB). Use the load-adjusted clock checks.py already has for budgets.
  • Not a CI lane yet. Add a pytest for the generator on a tiny N.

The design note holds the prototype generator used for the plan's Measurements table (75-line synthetic cards, three variants); replace its synthetic template with clones of real files.

Acceptance criteria

just scale --cards 2000 --crates 4 prints the four measurements from a workspace outside the tree; pytest covers the generator; just check passes.

Design

Prototype generator from the planning session (scratch, never committed):

"""Generate a synthetic crate of N Counterspell-shaped cards to measure rustc scaling.

variant: fn    = today's style (generic fn body, bodies!, scenarios in file)
         data  = no fn body (shared engine body), scenarios kept
         bare  = no fn body, no scenarios
"""
import pathlib, sys

variant, n, shards = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
root = pathlib.Path(sys.argv[4])
BERYL = "/Users/dthurn/beryl/rust/crates"

HEAD = """use beryl_engine::card::{{
    Ability, AbilityKind, CardDef, Colors, Face, Flow, ManaCost, TargetKind, TargetSpec, Types,
}};
use beryl_engine::effect::EffectCtx;
use beryl_engine::filter::Filter;
use beryl_engine::{{DecisionKind, Seat, TurnStep, Zone}};
use beryl_engine::{{Layout, bodies}};
use beryl_scenario::{{Action, Assertion, P0, P1, Permanent, Pick, Scenario}};

pub static CARD: CardDef = CardDef {{
    name: "Card {i}",
    faces: &[Face {{
        cost: ManaCost::new({g}, [0, 2, 0, 0, 0, 0]),
        types: Types::INSTANT,
        colors: Colors::BLUE,
        abilities: &[Ability {{
            kind: AbilityKind::Spell,
            targets: &[TargetSpec::one(TargetKind::Spell, Filter::Any)],
            body: bodies!({body}),
            ..Ability::NONE
        }}],
        ..Face::EMPTY
    }}],
    tokens: &[],
}};
"""
BODY = """
fn counter<L: Layout>(ctx: &mut EffectCtx<'_, L>) -> Flow {{
    if let Some(spell) = ctx.target_object(0) {{
        ctx.counter(spell, false);
    }}
    ctx.draw(ctx.controller(), {d});
    ctx.gain_life(ctx.controller(), {g});
    Ok(())
}}
"""
SCEN = """
const ISLANDS: &[Permanent] = &[Permanent::new("Island"), Permanent::new("Island")];

const fn count(player: Seat, zone: Zone, card: &'static str, count: u8) -> Action {{
    Action::assert(Assertion::Count {{ player, zone, card: Some(card), count }})
}}

pub const SCENARIOS: &[Scenario] = &[
    Scenario::new("Card {i} counters a spell into its owner's graveyard")
        .battlefield(P0, ISLANDS)
        .hand(P0, &["Card {i}"])
        .battlefield(P1, &[Permanent::new("Plains")])
        .hand(P1, &["Mana Crypt", "Plains"])
        .library(P0, &["Island"])
        .at(2, P1, TurnStep::PrecombatMain)
        .actions(&[
            Action::cast(P1, "Mana Crypt"),
            Action::cast(P0, "Card {i}"),
            count(P1, Zone::Graveyard, "Mana Crypt", 1),
            count(P1, Zone::Battlefield, "Mana Crypt", 0),
            count(P0, Zone::Graveyard, "Card {i}", 1),
        ]),
    Scenario::new("Card {i} chooses among spells on the stack")
        .battlefield(P0, ISLANDS)
        .hand(P0, &["Mana Crypt", "Card {i}", "Plains"])
        .battlefield(P1, &[Permanent::new("Plains")])
        .hand(P1, &["Mana Tithe"])
        .library(P1, &["Island"])
        .actions(&[
            Action::cast(P0, "Mana Crypt"),
            Action::pass(P0),
            Action::cast(P1, "Mana Tithe"),
            Action::cast(P0, "Card {i}"),
            Action::answer(P0, DecisionKind::Target, Pick::Card {{ card: "Mana Tithe" }}),
            Action::Resolve,
            count(P1, Zone::Graveyard, "Mana Tithe", 1),
            Action::Resolve,
            count(P0, Zone::Battlefield, "Mana Crypt", 1),
        ]),
];
"""

def card(i):
    g, d = i % 7, 1 + i % 3
    body = "counter" if variant == "fn" else "beryl_engine::effect::draw_a_card"
    s = "#![allow(unused_imports, clippy::all)]\n" + HEAD.format(i=i, g=g, body=body)
    if variant == "fn":
        s += BODY.format(d=d, g=g)
    if variant != "bare":
        s += SCEN.format(i=i)
    return s

members = []
per = n // shards
for sh in range(shards):
    c = root / f"shard{sh}"
    (c / "src").mkdir(parents=True, exist_ok=True)
    mods = []
    for k in range(per):
        i = sh * per + k
        (c / "src" / f"card_{i}.rs").write_text(card(i))
        mods.append(i)
    lib = "".join(f"pub mod card_{i};\n" for i in mods)
    lib += "pub static CARDS: &[&beryl_engine::card::CardDef] = &[\n" + "".join(f"    &card_{i}::CARD,\n" for i in mods) + "];\n"
    if variant != "bare":
        lib += "pub static SCENARIOS: &[&[beryl_scenario::Scenario]] = &[\n" + "".join(f"    card_{i}::SCENARIOS,\n" for i in mods) + "];\n"
    (c / "src" / "lib.rs").write_text(lib)
    (c / "Cargo.toml").write_text(f"""[package]
name = "shard{sh}"
version = "0.0.0"
edition = "2024"
[dependencies]
beryl_engine = {{ path = "{BERYL}/beryl_engine" }}
beryl_scenario = {{ path = "{BERYL}/beryl_scenario" }}
""")
    members.append(f"shard{sh}")
# a final binary that links every shard
b = root / "all"
(b / "src").mkdir(parents=True, exist_ok=True)
(b / "src" / "main.rs").write_text("fn main() {\n    let mut n = 0;\n" + "".join(f"    n += {m}::CARDS.len();\n" for m in members) + "    println!(\"{n}\");\n}\n")
(b / "Cargo.toml").write_text("[package]\nname = \"all\"\nversion = \"0.0.0\"\nedition = \"2024\"\n[dependencies]\n" + "".join(f"{m} = {{ path = \"../{m}\" }}\n" for m in members))
(root / "Cargo.toml").write_text("""[workspace]
resolver = "3"
members = ["all", %s]
[profile.release]
debug = "line-tables-only"
codegen-units = 1
lto = "thin"
""" % ", ".join(f'"{m}"' for m in members))
print("generated", n, variant, shards, "lines/card", card(0).count("\n"))

Handoff: tools/src/beryl_tools/scale.py (+ tests/test_scale.py, cli.py and justfile wiring). just scale --cards N --crates K (defaults 50,000 / 20) generates $BERYL_CACHE/scale (keeps target/), warms beryl_cards in check and release, then times cold check, release build of the all binary, and a one-card rebuild, reporting peak RSS from /usr/bin/time -l and load-adjusted time via checks.Clock. Decisions: pool is the 41 real card files reaching only crate::names and crate::tokens (others use super:: or pub(crate) helpers of siblings, e.g. timetwister::wheel, so they cannot compile outside beryl_cards); each shard re-exports beryl_cards::tokens and has its own names module (real names plus one const per clone); one-card rebuild is a cargo check one (edit-and-look loop); release profile copies the checkout's. Cargo.lock is copied from rust/ for pinned deps. Measured on a heavily loaded machine (load 60): 2,000 cards/4 crates: check 5.4 s, release 13.6 s, rebuild 2.5 s, peak 0.70 GiB. Milestone 7's --assert is not added.
